\node[body] (s1) at (0*\dx,0) {%
  \textbf{Inputs}\\ MCPTox, MCP-SafetyBench, AgentDojo, InjecAgent, A2A sandbox\\[3pt]
  \textbf{Rendering}\\ each case $\rightarrow$ episode $E=(m_{1},\dots,m_{T})$ of JSON-RPC messages; label $y\in\{0,1\}$, $1$ = attack\\[3pt]
  \textbf{Message tuple}\\ $m=(\tau,s,d,x,\mu,t)$: type, source, destination, payload, metadata, time};
\node[body] (s2) at (1*\dx,0) {%
  \textbf{Grouped splits}\\ no server, case, template or twin crosses splits:\\ train $\mid$ val$_{\text{sel}}$ $\mid$ val$_{\text{cal}}$ $\mid$ test\\[3pt]
  \textbf{Held-out folds $\calF$}\\ hide one wording, family or vulnerability class from train and val; test on it alone\\[3pt]
  \textbf{Transfer}\\ train on one benchmark, test on another\\[3pt]
  \textbf{Audit}\\ leakage check per fold; length control};
\node[body] (s3) at (2*\dx,0) {%
  \textbf{Chunking}\\ payload $x\rightarrow c_{1},\dots,c_{K}$: one per tool, per ${\sim}48$-word sentence run, plus a header ($K\le64$)\\[3pt]
  \textbf{Per chunk}\\ $e(c_{k})\in\R^{384}$: MiniLM\\ $v(c_{k})\in\R^{512}$: hashed word counts\\[3pt]
  \textbf{Pooling}\\ $p=[\overline{e},\max e,\max v]$\\ learned scorer $g(c_{k})$, max + mean};
\node[body] (s4) at (3*\dx,0) {%
  \textbf{Controls}\\ length; lexical TF--IDF + logistic regression; novelty (Mahalanobis)\\[3pt]
  \textbf{Independent}\\ ProtectAI, PIGuard (zero-shot); LLM judge; fine-tuned DeBERTa-v3\\[3pt]
  \textbf{MambaGuard}\\ $f_{\theta}$: SSM + GATv2, graph or flat read-out; 5 ablations\\[3pt]
  each outputs $s(E)\in[0,1]$};
\node[body] (s5) at (4*\dx,0) {%
  \textbf{Calibration}\\ Platt: $\tilde s=\sigma(a s+b)$, fitted on val$_{\text{sel}}$\\[3pt]
  \textbf{Hybrid}\\ $s_{\text{hyb}}=\sum_{j}w_{j}\tilde s_{j}$; weights $w$ on the simplex, chosen on val$_{\text{sel}}$\\[3pt]
  \textbf{Alarm threshold}\\ $\hat\tau$: best val$_{\text{sel}}$ macro-F1\\ $\tau_{\alpha}$: split conformal on val$_{\text{cal}}$\\ $\alpha_{t}$: adaptive conformal (ACI)};
\node[body] (s6) at (5*\dx,0) {%
  \textbf{Metrics}\\ macro-F1, AUROC, recall, FPR, ECE\\[3pt]
  \textbf{Statistics}\\ 5 seeds; paired bootstrap with Holm correction; sign test over folds\\[3pt]
  \textbf{Guarantees}\\ smoothing $\mathrm{CertAcc}(r)$; radius ceiling; Lipschitz audit\\[3pt]
  \textbf{Cost}\\ latency and throughput};
